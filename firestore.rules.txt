rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
  function signedIn(){return request.auth != null;}
  function isAdmin(){return signedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid)) && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';}
  function isOwner(){return signedIn() && resource.data.ownerUid == request.auth.uid;}
  function createsOwnDocument(){return signedIn() && request.resource.data.ownerUid == request.auth.uid;}
  function updatesOwnDocument(){return isOwner() && request.resource.data.ownerUid == resource.data.ownerUid;}
  match /users/{userId}{allow read: if (signedIn() && request.auth.uid == userId) || isAdmin(); allow write: if false;}
  match /carePlans/{id}{allow read: if signedIn(); allow create: if createsOwnDocument(); allow update: if updatesOwnDocument(); allow delete: if isOwner() || isAdmin();}
  match /counsels/{id}{allow read: if signedIn(); allow create: if createsOwnDocument(); allow update: if updatesOwnDocument(); allow delete: if isOwner() || isAdmin();}
  match /attendance/{id}{allow read: if signedIn(); allow create: if createsOwnDocument(); allow update: if updatesOwnDocument(); allow delete: if isOwner() || isAdmin();}
  match /outings/{id}{allow read: if signedIn(); allow create: if createsOwnDocument(); allow update: if updatesOwnDocument(); allow delete: if isOwner() || isAdmin();}
  match /attendanceMonthLocks/{month}{allow read: if signedIn(); allow create: if createsOwnDocument() && request.resource.data.month == month; allow update: if false; allow delete: if isOwner() || isAdmin();}
  match /portalSettings/{id}{allow read: if signedIn(); allow create, update, delete: if isAdmin();}
  match /needsAssessmentSettings/{id}{allow read: if signedIn(); allow create, update, delete: if isAdmin();}
  match /dailyTasks/{taskId}{
   allow read: if signedIn() && (isAdmin() || resource.data.assigneeUid == request.auth.uid);
   allow create, delete: if isAdmin();
   allow update: if isAdmin() || (signedIn() && resource.data.assigneeUid == request.auth.uid && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['completed','completedAt']) && request.resource.data.completed is bool);
  }
  // 등록된 업무는 관리자만 변경 가능
  match /taskTemplates/{id}{allow read: if signedIn(); allow create, update, delete: if isAdmin();}
  // 직원은 본인에게 배정된 업무만 조회 가능
  match /taskAssignments/{id}{allow read: if signedIn(); allow create, update, delete: if isAdmin();}
  // 직원은 본인 업무의 완료 여부만 기록 가능
  match /staffSchedules/{id}{
    allow read: if signedIn();
    allow create, update, delete: if isAdmin();
  }
  match /taskHandoffs/{id}{
    allow read: if signedIn();
    allow create: if signedIn()
      && request.resource.data.fromUid == request.auth.uid
      && request.resource.data.toUid != request.auth.uid
      && request.resource.data.status == 'pending'
      && request.resource.data.keys().hasOnly(['assignmentId','period','fromUid','toUid','reason','status','createdAt'])
      && exists(/databases/$(database)/documents/taskAssignments/$(request.resource.data.assignmentId));
    allow update: if isAdmin() || (signedIn()
      && resource.data.toUid == request.auth.uid
      && resource.data.status == 'pending'
      && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','respondedAt'])
      && request.resource.data.status in ['accepted','rejected']);
    allow delete: if isAdmin();
  }
  match /taskCompletions/{id}{
    allow read: if signedIn();
    allow create: if signedIn()
      && request.resource.data.assigneeUid == request.auth.uid
      && request.resource.data.completed == true
      && request.resource.data.keys().hasOnly(['assignmentId','period','assigneeUid','completed','completedAt'])
      && exists(/databases/$(database)/documents/taskAssignments/$(request.resource.data.assignmentId))
      && id == request.resource.data.assignmentId + '_' + request.resource.data.period;
    allow update: if false;
    allow delete: if isAdmin() || (signedIn() && resource.data.assigneeUid == request.auth.uid);
  }
  // 직원 본인만 개인 업무를 생성/수정/삭제. 관리자도 타인의 개인 업무 수정 불가.
  match /personalTasks/{id}{
    allow read: if signedIn();
    allow create: if signedIn() && request.resource.data.ownerUid == request.auth.uid
      && request.resource.data.date is string && request.resource.data.title is string
      && request.resource.data.completed is bool;
    allow update: if signedIn() && resource.data.ownerUid == request.auth.uid
      && request.resource.data.ownerUid == resource.data.ownerUid
      && request.resource.data.date == resource.data.date;
    allow delete: if signedIn() && resource.data.ownerUid == request.auth.uid;
  }
  // 배정 업무 및 개인 업무의 시간 배치는 각자 본인만 변경.
  match /taskTimePlans/{id}{
    allow read: if signedIn();
    allow create: if signedIn() && request.resource.data.ownerUid == request.auth.uid
      && request.resource.data.date is string && request.resource.data.taskKey is string
      && request.resource.data.start is string && request.resource.data.duration is int;
    allow update: if signedIn() && resource.data.ownerUid == request.auth.uid
      && request.resource.data.ownerUid == resource.data.ownerUid
      && request.resource.data.date == resource.data.date
      && request.resource.data.taskKey == resource.data.taskKey;
    allow delete: if signedIn() && resource.data.ownerUid == request.auth.uid;
  }
  match /{document=**}{allow read, write: if false;}
 }
}
